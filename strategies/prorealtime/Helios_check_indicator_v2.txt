// =====================================================================
// HELIOS check indicator v2 (ProBuilder indicator, DAILY chart, own panel)
// Uses exactly the weekday-session method of the v2 systems. Needs DEFAULT candle times
// (Data > Time zones & trading hours), so that IG's weekend bar is a short bar that opens on Sunday. Put it on a chart WITH
// weekend data shown: it should give the same numbers as the research (and as the old
// Claude Indicator with weekend data hidden).
// Expected for FTSE 100, Fri 20 Nov 2026: score about 0.15 (chart without weekends 0.155,
// research 0.151), lower band about 10,414, ATR about 185 (research 185.7).
// On weekend bars it repeats the last weekday values.
// =====================================================================
ONCE nS = 0
ONCE wkH = 0
ONCE wkL = 0
ONCE haveWk = 0
ONCE score = 0
ONCE lower = 0
ONCE atr = 0
ONCE er = 0

weekend = (OpenDayOfWeek = 0 OR OpenDayOfWeek = 6)   // the bar OPENED on a Saturday or Sunday (IG dates bars by their close, so DayOfWeek cannot be used)

IF weekend THEN
   IF haveWk = 0 THEN
      wkH = high
      wkL = low
      haveWk = 1
   ELSE
      wkH = MAX(wkH, high)
      wkL = MIN(wkL, low)
   ENDIF
ELSE
   hiS = high
   loS = low
   IF haveWk = 1 THEN
      hiS = MAX(hiS, wkH)
      loS = MIN(loS, wkL)
      haveWk = 0
   ENDIF
   $cl[nS] = close
   $hi[nS] = hiS
   $lo[nS] = loS
   nS = nS + 1
   IF nS >= 122 THEN
      lastI = nS - 1
      s0 = 0
      s20 = 0
      FOR ii = 0 TO 99 DO
         s0 = s0 + $cl[lastI - ii]
         s20 = s20 + $cl[lastI - 20 - ii]
      NEXT
      slope = (s0 / s20 - 1) * 100
      sr = 0
      FOR ii = 0 TO 99 DO
         sr = sr + LOG($cl[lastI - ii] / $cl[lastI - ii - 1])
      NEXT
      mr = sr / 100
      ss = 0
      FOR ii = 0 TO 99 DO
         dv = LOG($cl[lastI - ii] / $cl[lastI - ii - 1]) - mr
         ss = ss + dv * dv
      NEXT
      volS = SQRT(ss / 100) * SQRT(20) * 100
      IF volS > 0 THEN
         score = slope / volS
      ENDIF
      sm = 0
      FOR ii = 0 TO 19 DO
         sm = sm + $cl[lastI - ii]
      NEXT
      mid = sm / 20
      sv = 0
      FOR ii = 0 TO 19 DO
         sv = sv + ($cl[lastI - ii] - mid) * ($cl[lastI - ii] - mid)
      NEXT
      lower = mid - 2 * SQRT(sv / 20)
      st = 0
      FOR ii = 0 TO 13 DO
         kk = lastI - ii
         st = st + MAX($hi[kk] - $lo[kk], MAX(ABS($hi[kk] - $cl[kk - 1]), ABS($lo[kk] - $cl[kk - 1])))
      NEXT
      atr = st / 14
      path = 0
      FOR ii = 0 TO 19 DO
         path = path + ABS($cl[lastI - ii] - $cl[lastI - ii - 1])
      NEXT
      IF path > 0 THEN
         er = ($cl[lastI] - $cl[lastI - 20]) / path
      ENDIF
   ENDIF
ENDIF

RETURN score AS "H1a trend score (weekdays)", 0.15 AS "threshold", lower AS "lower band (weekdays)", atr AS "ATR14 (weekdays)", er AS "ER20 (weekdays)"
