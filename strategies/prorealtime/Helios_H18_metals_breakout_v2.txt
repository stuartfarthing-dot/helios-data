// =====================================================================
// HELIOS H18a v2 - metals breakout                   (ProOrder, DAILY chart)
// v2 (23 Nov 2026): IGNORES WEEKEND BARS (see the H1a v2 notes). The 252-session high
// and ATR are calculated from the system's own list of weekday sessions; a weekend bar's
// high and low are folded into Monday. No signal on a weekend bar. Rule unchanged
// (pre-registered 10 Nov 2026). Spot Gold (DFB) and Spot Silver (DFB) only.
// Signal: the first daily close at a 252-session closing high, with no such close in the
// previous 20 sessions. Buy at the next open. Stop 2 x ATR(14).
// Exit C2: trailing stop 2 x ATR (at the signal) below the highest high since entry,
// never lowered; or after 100 sessions.
// =====================================================================
DEFPARAM CumulateOrders = False
DEFPARAM PreLoadBars = 800           // about 3 years of bars, enough for 273 weekday sessions

// ---------------- settings ----------------
PositionSize = 0.01                  // GBP per point
// ------------------------------------------

ONCE nS = 0
ONCE wkH = 0
ONCE wkL = 0
ONCE haveWk = 0
ONCE riskdist = 0
ONCE entryN = 0
ONCE hh = 0

weekend = (OpenDayOfWeek = 0 OR OpenDayOfWeek = 6)   // the bar OPENED on a Saturday or Sunday (IG dates bars by their close, so DayOfWeek cannot be used)
newsession = 0

IF weekend THEN
   IF haveWk = 0 THEN
      wkH = high
      wkL = low
      haveWk = 1
   ELSE
      wkH = MAX(wkH, high)
      wkL = MIN(wkL, low)
   ENDIF
ELSE
   hiS = high
   loS = low
   IF haveWk = 1 THEN
      hiS = MAX(hiS, wkH)
      loS = MIN(loS, wkL)
      haveWk = 0
   ENDIF
   $cl[nS] = close
   $hi[nS] = hiS
   $lo[nS] = loS
   nS = nS + 1
   newsession = 1
ENDIF

setup = 0
IF newsession = 1 AND nS >= 252 THEN
   lastI = nS - 1
   hi = $cl[lastI]
   FOR ii = 1 TO 251 DO
      hi = MAX(hi, $cl[lastI - ii])
   NEXT
   IF $cl[lastI] >= hi THEN
      $nhh[lastI] = 1
   ELSE
      $nhh[lastI] = 0
   ENDIF
   IF nS >= 273 AND $nhh[lastI] = 1 THEN
      recent = 0
      FOR ii = 1 TO 20 DO
         IF ISSET($nhh[lastI - ii]) THEN
            recent = recent + $nhh[lastI - ii]
         ELSE
            recent = recent + 1
         ENDIF
      NEXT
      IF recent = 0 THEN
         setup = 1
      ENDIF
   ENDIF
   st = 0
   FOR ii = 0 TO 13 DO
      kk = lastI - ii
      st = st + MAX($hi[kk] - $lo[kk], MAX(ABS($hi[kk] - $cl[kk - 1]), ABS($lo[kk] - $cl[kk - 1])))
   NEXT
   atr = st / 14
ENDIF

// ---------------- entry ----------------
IF NOT ONMARKET AND setup = 1 THEN
   riskdist = 2 * atr
   entryN = nS
   hh = 0
   BUY PositionSize PERPOINT AT MARKET
   SET STOP LOSS riskdist
ENDIF

// ---------------- management (C2) ----------------
IF ONMARKET THEN
   IF hh = 0 THEN
      hh = high
   ELSE
      hh = MAX(hh, high)
   ENDIF
   barsheld = nS - entryN
   entryprice = TRADEPRICE(1)
   trail = MAX(entryprice - riskdist, hh - riskdist)
   IF newsession = 1 AND barsheld >= 100 THEN
      SELL AT MARKET
   ELSE
      SELL AT trail STOP
   ENDIF
ENDIF
