// =====================================================================
// HELIOS H14 v2 - buy persistent selling             (ProOrder, DAILY chart)
// v2 (23 Nov 2026): IGNORES WEEKEND BARS (see the H1a v2 notes). Every measure is
// calculated from the system's own list of weekday sessions; a weekend bar's high and
// low are folded into Monday. No signal or exit decision on a weekend bar.
// The RULE is unchanged (pre-registered 10 Nov 2026).
// Markets: FTSE 100, US 500, US Tech 100, Wall Street, Germany 40, Australia 200,
// Spot Gold, Spot Silver, Platinum; Spain 35 and France 40 (added 17 Nov).
// Signal: the FIRST session in "persistent selling": 20-session efficiency ratio <= -0.35,
// not a panic day and not a euphoria day (H13 states). Buy at the next open.
// Stop 2 x ATR(14). Exit B: first close above the 20-session middle band, or 20 sessions.
// =====================================================================
DEFPARAM CumulateOrders = False
DEFPARAM PreLoadBars = 400

// ---------------- settings ----------------
PositionSize = 0.01                  // GBP per point
// ------------------------------------------

ONCE nS = 0
ONCE wkH = 0
ONCE wkL = 0
ONCE haveWk = 0
ONCE streak = 0
ONCE mid = 0
ONCE atr = 0
ONCE riskdist = 0
ONCE entryN = 0

weekend = (OpenDayOfWeek = 0 OR OpenDayOfWeek = 6)   // the bar OPENED on a Saturday or Sunday (IG dates bars by their close, so DayOfWeek cannot be used)
newsession = 0

IF weekend THEN
   IF haveWk = 0 THEN
      wkH = high
      wkL = low
      haveWk = 1
   ELSE
      wkH = MAX(wkH, high)
      wkL = MIN(wkL, low)
   ENDIF
ELSE
   hiS = high
   loS = low
   IF haveWk = 1 THEN
      hiS = MAX(hiS, wkH)
      loS = MIN(loS, wkL)
      haveWk = 0
   ENDIF
   $cl[nS] = close
   $hi[nS] = hiS
   $lo[nS] = loS
   nS = nS + 1
   newsession = 1
ENDIF

setup = 0
IF newsession = 1 AND nS >= 40 THEN
   lastI = nS - 1
   // Bollinger (20, 2) and z-score
   sm = 0
   FOR ii = 0 TO 19 DO
      sm = sm + $cl[lastI - ii]
   NEXT
   mid = sm / 20
   sv = 0
   FOR ii = 0 TO 19 DO
      sv = sv + ($cl[lastI - ii] - mid) * ($cl[lastI - ii] - mid)
   NEXT
   sdS = SQRT(sv / 20)
   IF sdS > 0 THEN
      bbz = ($cl[lastI] - mid) / sdS
   ELSE
      bbz = 0
   ENDIF
   $zz[lastI] = bbz
   // 20-session efficiency ratio
   path = 0
   FOR ii = 0 TO 19 DO
      path = path + ABS($cl[lastI - ii] - $cl[lastI - ii - 1])
   NEXT
   IF path > 0 THEN
      er = ($cl[lastI] - $cl[lastI - 20]) / path
   ELSE
      er = 0
   ENDIF
   // true range today and the average of the previous 14 sessions
   trnow = MAX($hi[lastI] - $lo[lastI], MAX(ABS($hi[lastI] - $cl[lastI - 1]), ABS($lo[lastI] - $cl[lastI - 1])))
   sp = 0
   FOR ii = 1 TO 14 DO
      kk = lastI - ii
      sp = sp + MAX($hi[kk] - $lo[kk], MAX(ABS($hi[kk] - $cl[kk - 1]), ABS($lo[kk] - $cl[kk - 1])))
   NEXT
   atrprev = sp / 14
   IF atrprev > 0 THEN
      shock = trnow / atrprev
   ELSE
      shock = 0
   ENDIF
   // ATR as in the backtest: simple average of the last 14 true ranges (including today)
   atr = (sp - MAX($hi[lastI - 14] - $lo[lastI - 14], MAX(ABS($hi[lastI - 14] - $cl[lastI - 15]), ABS($lo[lastI - 14] - $cl[lastI - 15]))) + trnow) / 14
   // consecutive down closes
   IF $cl[lastI] < $cl[lastI - 1] THEN
      streak = streak + 1
   ELSE
      streak = 0
   ENDIF
   // closes at or above +1.5 SD in the last 10 sessions
   upperdays = 0
   FOR ii = 0 TO 9 DO
      IF ISSET($zz[lastI - ii]) THEN
         IF $zz[lastI - ii] >= 1.5 THEN
            upperdays = upperdays + 1
         ENDIF
      ENDIF
   NEXT
   panic = (bbz <= -2 AND (shock >= 2 OR streak >= 4))
   euphoria = (bbz >= 2 AND upperdays >= 4)
   IF (NOT panic) AND (NOT euphoria) AND er <= -0.35 THEN
      $pss[lastI] = 1
   ELSE
      $pss[lastI] = 0
   ENDIF
   prevps = 0
   IF ISSET($pss[lastI - 1]) THEN
      prevps = $pss[lastI - 1]
   ENDIF
   IF $pss[lastI] = 1 AND prevps = 0 THEN
      setup = 1
   ENDIF
ENDIF

// ---------------- entry ----------------
IF NOT ONMARKET AND setup = 1 THEN
   riskdist = 2 * atr
   entryN = nS
   BUY PositionSize PERPOINT AT MARKET
   SET STOP LOSS riskdist
ENDIF

// ---------------- exit B ----------------
IF ONMARKET AND newsession = 1 THEN
   barsheld = nS - entryN
   IF $cl[nS - 1] > mid OR barsheld >= 20 THEN
      SELL AT MARKET
   ENDIF
ENDIF
